import torch
import torch.nn as nn
from torch.nn.utils import parametrize

class RefusalConstraint(nn.Module):
    def __init__(self, refusal_map_tensor):
        super().__init__()
        # refusal_map_tensor: [N_experts, Latent_Dim] or similar coordinate representation
        self.register_buffer("refusal_map", refusal_map_tensor)
        self.penalty_coefficient = 1e6

    def forward(self, routing_logits, hidden_states):
        # Calculate distance from current activation to refusal manifold
        # This is a simplified distance metric for the prototype
        dist_to_refusal = torch.norm(hidden_states.unsqueeze(1) - self.refusal_map.unsqueeze(0), dim=-1)
        
        # Apply penalty to logits for paths approaching refusal zones
        penalty = torch.exp(-dist_to_refusal) * self.penalty_coefficient
        return routing_logits - penalty
